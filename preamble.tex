% ===== ElegantBook customizations for 《vLLM 学习手册》 =====
% 借鉴 ai-agent-book 的 preamble 设计，适配 vLLM 推理工程主题。
% Accent color: ElegantBook green theme 的深绿，搭配酒红点缀。

% ── Microtype: disable protrusion ────────────────────────────────
% ElegantBook loads microtype, but its protrusion pass trips over
% \newunicodechar-defined glyphs (extra } in \MT@is@char). Disabling
% protrusion keeps tracking/expansion (the visually useful parts) while
% avoiding the conflict.
\microtypesetup{protrusion=false}

% ── Counter format: use arabic for sub-levels ────────────────────
% ElegantBook's default uses \alph for subsubsection-level theorem
% counters, which overflows past 26. Force arabic to handle 60-chapter
% books with deep numbering.
\renewcommand{\thesubsubsection}{\arabic{subsubsection}}

% ── Front matter page numbering: roman instead of Alph ───────────
% ElegantBook's \maketitle sets \pagenumbering{Alph}, which overflows
% past 26 pages (our front matter is ~25 pages of TOC + title). Patch
% \maketitle to restore roman numbering right after the title page.
\AtBeginDocument{%
  \let\origmaketitle\maketitle
  \renewcommand{\maketitle}{\origmaketitle\pagenumbering{roman}}%
}

% ── tcolorbox (ElegantBook already loads it; add libraries) ──────
\tcbuselibrary{skins,breakable}

% ── TikZ (for the cover) ─────────────────────────────────────────
\usepackage{tikz}
\usetikzlibrary{fadings}

% Theme accent color: deep green (classic technical-book tone).
% Drives headings, rules, links, code chrome and the cover.
\definecolor{structurecolor}{RGB}{30,99,65}

% Accent shade for code/figure chrome
\colorlet{accent}{structurecolor}
\colorlet{accentbg}{structurecolor!6}
\colorlet{accentrule}{structurecolor!35}

% ── Wrap long lines in code blocks ───────────────────────────────
\usepackage{fvextra}
\fvset{breaklines=true,breakanywhere=true,%
  breaksymbolleft={\footnotesize\textcolor{accentrule}{$\hookrightarrow$}\,}}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},breaklines,breakanywhere}

% ── Code blocks: subtle framed box ───────────────────────────────
\newtcolorbox{codebox}{%
  breakable, enhanced,
  colback=black!3, colframe=accentrule,
  boxrule=0.5pt, leftrule=2.5pt,
  left=0.7em, right=0.6em, top=0.4em, bottom=0.4em,
  arc=0.6mm,
  before skip=0.7em, after skip=0.7em
}
\renewenvironment{Shaded}{\begin{codebox}}{\end{codebox}}
\renewenvironment{verbatim}%
  {\VerbatimEnvironment\begin{codebox}\begin{Verbatim}}%
  {\end{Verbatim}\end{codebox}}

% ── Figures: force [H] ────────────────────────────────────────────
\usepackage{float}
\let\origfigure\figure
\let\origendfigure\endfigure
\renewenvironment{figure}[1][htbp]{\origfigure[H]}{\origendfigure}
\setkeys{Gin}{width=\linewidth,height=0.42\textheight,keepaspectratio}

% ── Figure captions ──────────────────────────────────────────────
\usepackage{caption}
\IfFontExistsTF{Heiti SC}{\setCJKfamilyfont{capcjksf}[AutoFakeBold=2.5]{Heiti SC}}{\IfFontExistsTF{Noto Sans CJK SC}{\setCJKfamilyfont{capcjksf}[AutoFakeBold=2.5]{Noto Sans CJK SC}}{\setCJKfamilyfont{capcjksf}[AutoFakeBold=2.5]{DejaVu Sans}}}
\DeclareCaptionFont{cjkmatch}{\CJKfamily{capcjksf}}
\captionsetup{font={normalsize,sf,bf,cjkmatch}, labelformat=empty, skip=4pt, justification=centering}

% ── Blockquotes: accent left bar ─────────────────────────────────

    before skip=0.6em, after skip=0.6em
  ]%
}{%
  \end{tcolorbox}%
}
